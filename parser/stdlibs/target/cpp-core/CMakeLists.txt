cmake_minimum_required(VERSION 3.22)
project(ispa_cpp_core LANGUAGES CXX)

# ==============================================================
# ispastdlib: the hand-written C++ runtime. ispa_core (below) is
# the facade that bridges it to the lowered core; both live here
# together now so cpp-core installs and builds as one self-contained
# package (ispa-pkg install cpp-core needs nothing from elsewhere
# but target-core).
# ==============================================================
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

file(GLOB_RECURSE ISPA_STDLIB_HEADERS
        CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/ispastdlib.hpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/ispastdlib/*.hpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/ispastdlib/*.h"
)

file(GLOB ISPA_STDLIB_SOURCES
        CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/src/core.cpp"
)

# OS headers define macros such as IN that collide with generated token names.
# Only core.cpp may include them; never inject them through the runtime PCH.
list(REMOVE_ITEM ISPA_STDLIB_HEADERS
        "${CMAKE_CURRENT_SOURCE_DIR}/ispastdlib/platform.hpp"
)

if(ISPA_STDLIB_SOURCES)
    add_library(ispastdlib STATIC ${ISPA_STDLIB_SOURCES})
else()
    add_library(ispastdlib INTERFACE)
endif()
add_library(ispastdlib::ispastdlib ALIAS ispastdlib)

# ispastdlib_deps.json/.inc and ispaext_deps.inc are a scan of the headers above (tools/ispa_deps.py),
# not source: regenerate them on every configure instead of committing stale output to git.
# lib/converter/cpp/converter.cpp reads them through tools/LibraryDeps.hpp (constexpr #include).
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_FOUND)
    execute_process(
            COMMAND "${Python3_EXECUTABLE}" tools/ispa_deps.py
                    --root "${CMAKE_CURRENT_SOURCE_DIR}"
                    --out "${CMAKE_CURRENT_SOURCE_DIR}/ispastdlib_deps.json"
                    --emit-inc "${CMAKE_CURRENT_SOURCE_DIR}/ispastdlib_deps.inc"
            WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
            RESULT_VARIABLE ISPA_DEPS_SCAN_RESULT
    )
    if(NOT ISPA_DEPS_SCAN_RESULT EQUAL 0)
        message(FATAL_ERROR "cpp-core: tools/ispa_deps.py failed to scan ispastdlib/ (exit ${ISPA_DEPS_SCAN_RESULT})")
    endif()
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/ispaext_deps.json")
        file(WRITE "${CMAKE_CURRENT_SOURCE_DIR}/ispaext_deps.json" "{}")
    endif()
    execute_process(
            COMMAND "${Python3_EXECUTABLE}" tools/gen_ext_deps.py
            WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
            RESULT_VARIABLE ISPA_EXT_DEPS_RESULT
    )
    if(NOT ISPA_EXT_DEPS_RESULT EQUAL 0)
        message(FATAL_ERROR "cpp-core: tools/gen_ext_deps.py failed (exit ${ISPA_EXT_DEPS_RESULT})")
    endif()
elseif(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/ispastdlib_deps.inc")
    message(FATAL_ERROR "cpp-core: Python3 not found and ispastdlib_deps.inc is not generated; install Python3 to regenerate the dependency manifest")
endif()

if(TARGET ispastdlib)
    target_include_directories(ispastdlib PUBLIC
            $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
            $<INSTALL_INTERFACE:include>
    )
    target_compile_features(ispastdlib PUBLIC cxx_std_17)

    # Consumers may import std; injecting this PCH mixes textual standard
    # library declarations with the module's declarations (notably on Clang 20).
    target_precompile_headers(ispastdlib PRIVATE
            ${ISPA_STDLIB_HEADERS}
    )

    if(WIN32)
        # Keep min/max macros out of the stack-trace implementation and consumers
        # that include Windows headers themselves.
        target_compile_definitions(ispastdlib PUBLIC NOMINMAX)
        target_link_libraries(ispastdlib PRIVATE dbghelp)
    endif()

    if(MSVC)
        target_compile_options(ispastdlib PRIVATE /MP /bigobj)
    endif()
endif()

# ==============================================================
# ispa_core: the language-neutral runtime (target-core's ispa_std)
# lowered to C++ (ispa_core_lowered.{h,cpp}, regenerated by regen.py),
# linked against ispastdlib above through the facade headers in ispa_core/.
# ==============================================================
if(NOT TARGET ispa_core)
    add_library(ispa_core STATIC "${CMAKE_CURRENT_SOURCE_DIR}/src/ispa_core_lowered.cpp")
    target_include_directories(ispa_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_features(ispa_core PUBLIC cxx_std_17)
    target_link_libraries(ispa_core PUBLIC ispastdlib::ispastdlib)
endif()
